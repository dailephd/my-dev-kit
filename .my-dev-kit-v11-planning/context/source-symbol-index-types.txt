184 | /**
185 |  * The graph section embedded in the main symbol-index artifact.
186 |  *
187 |  * Provides compact, graph-oriented data derived from the per-file summaries:
188 |  * - resolved internal file dependency edges (suitable for graph traversal and
189 |  *   later file-dependency visualization)
190 |  * - a flat cross-file symbol lookup table (compact, no signatures)
191 |  *
192 |  * Relationship to other artifact parts:
193 |  * - `files[]`            — verbose per-file summaries with raw import specifiers
194 |  *                          and full symbol signatures (retrieval use)
195 |  * - `graph`              — resolved internal edges + compact cross-file symbol
196 |  *                          records (graph traversal / narrowing / visualization)
197 |  * - `callGraphArtifact`  — pointer to the separate, heavier call-graph artifact
198 |  *                          which carries caller?callee symbol-level edges
199 |  */
200 | export interface SymbolIndexGraph {
201 |   /**
202 |    * All resolved internal file-to-file dependency edges.
203 |    * Sorted by `from` then `to` for deterministic output.
204 |    * External/package imports are excluded.
205 |    */
206 |   fileDeps: FileDependencyEdge[]
207 |   /**
208 |    * Flat cross-file list of compact symbol records.
209 |    * Sorted by `file` then `name` for deterministic lookup.
210 |    * Use `FileSummary.symbols` when you also need `signature` text.
211 |    */
212 |   symbols: GraphSymbolRecord[]
213 | }
